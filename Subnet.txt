export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const action = url.searchParams.get('api_action');

    // CORS Headers برای پاسخ به درخواست‌های برنامه شما
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Content-Type": "application/json"
    };

    // ۱. هندل کردن API اعتبارسنجی (استفاده توسط نرم‌افزار شما)
    if (action === 'validate') {
      const key = url.searchParams.get('key');
      const hwid = url.searchParams.get('hwid');

      if (!key || !hwid) {
        return new Response(JSON.stringify({ status: 'ERROR', message: 'پارامترهای ورودی ناقص است' }), { headers: corsHeaders });
      }

      const licenseDataStr = await env.LICENSES_KV.get(key);
      if (!licenseDataStr) {
        return new Response(JSON.stringify({ status: 'INVALID', message: 'لایسنس یافت نشد' }), { headers: corsHeaders });
      }

      const license = JSON.parse(licenseDataStr);

      if (license.expire_date && new Date(license.expire_date) < new Date()) {
        return new Response(JSON.stringify({ status: 'EXPIRED', message: 'اعتبار لایسنس به پایان رسیده است' }), { headers: corsHeaders });
      }

      let devices = license.devices || [];
      if (devices.includes(hwid)) {
        return new Response(JSON.stringify({ status: 'SUCCESS', type: license.type, expires: license.expire_date }), { headers: corsHeaders });
      }

      if (devices.length >= license.max_devices) {
        return new Response(JSON.stringify({ status: 'MAX_DEVICES_REACHED', message: 'سقف تعداد دستگاه مجاز تکمیل شده است' }), { headers: corsHeaders });
      }

      devices.push(hwid);
      license.devices = devices;
      await env.LICENSES_KV.put(key, JSON.stringify(license));

      return new Response(JSON.stringify({ status: 'SUCCESS', type: license.type, expires: license.expire_date }), { headers: corsHeaders });
    }

    // بررسی کوکی یا وضعیت لاگین برای پنل مدیریت
    const cookie = request.headers.get('Cookie') || '';
    const isLoggedIn = cookie.includes('logged_in=true');

    // هندل کردن درخواست‌های POST پنل مدیریت
    if (request.method === 'POST') {
      const formData = await request.formData();
      const formAction = formData.get('action');

      if (formAction === 'login') {
        const user = formData.get('username');
        const pass = formData.get('password');
        if (user === 'خواجوی' && pass === 'M8056@8056m') {
          return new Response('', {
            status: 302,
            headers: {
              'Location': '/',
              'Set-Cookie': 'logged_in=true; Path=/; HttpOnly'
            }
          });
        } else {
          return new Response(getLoginPage("نام کاربری یا رمز عبور اشتباه است"), { headers: { "Content-Type": "text/html; charset=utf-8" } });
        }
      }

      if (isLoggedIn) {
        if (formAction === 'add_app') {
          const appName = formData.get('app_name')?.trim();
          if (appName) {
            let appsStr = await env.LICENSES_KV.get('__APPS__') || '[]';
            let apps = JSON.parse(appsStr);
            if (!apps.some(a => a.name === appName)) {
              apps.push({ id: Date.now().toString(), name: appName });
              await env.LICENSES_KV.put('__APPS__', JSON.stringify(apps));
            }
          }
        } else if (formAction === 'add_license') {
          const appId = formData.get('app_id');
          const type = formData.get('license_type');
          const maxDevices = parseInt(formData.get('max_devices') || '1');
          const expireDate = formData.get('expire_date') || null;

          const genKey = 'KEY-' + Math.random().toString(36).substring(2, 6).toUpperCase() + '-' + Math.random().toString(36).substring(2, 6).toUpperCase() + '-' + Math.random().toString(36).substring(2, 6).toUpperCase();
          
          const licenseData = {
            app_id: appId,
            type: type,
            max_devices: maxDevices,
            expire_date: expireDate,
            devices: [],
            created_at: newicoDate = new Date().toISOString().split('T')[0]
          };

          await env.LICENSES_KV.put(genKey, JSON.stringify(licenseData));
          
          let keysListStr = await env.LICENSES_KV.get(`__KEYS_${appId}__`) || '[]';
          let keysList = JSON.parse(keysListStr);
          keysList.unshift({ key: genKey, ...licenseData });
          await env.LICENSES_KV.put(`__KEYS_${appId}__`, JSON.stringify(keysList));
        }

        return new Response('', {
          status: 302,
          headers: { 'Location': url.pathname }
        });
      }
    }

    if (url.searchParams.get('logout') === 'true') {
      return new Response('', {
        status: 302,
        headers: {
          'Location': '/',
          'Set-Cookie': 'logged_in=false; Path=/; Max-Age=0'
        }
      });
    }

    // اگر کاربر لاگین نکرده است، صفحه لاگین را نشان بده
    if (!isLoggedIn) {
      return new Response(getLoginPage(), { headers: { "Content-Type": "text/html; charset=utf-8" } });
    }

    // بارگذاری اطلاعات برای پنل مدیریت
    let appsStr = await env.LICENSES_KV.get('__APPS__') || '[]';
    let apps = JSON.parse(appsStr);
    let selectedAppId = url.searchParams.get('app_id') || (apps[0] ? apps[0].id : null);
    
    let licenses = [];
    if (selectedAppId) {
      let keysListStr = await env.LICENSES_KV.get(`__KEYS_${selectedAppId}__`) || '[]';
      licenses = JSON.parse(keysListStr);
    }

    const baseUrl = `${url.protocol}//${url.host}${url.pathname}`;

    return new Response(getDashboardHTML(apps, selectedAppId, licenses, baseUrl), {
      headers: { "Content-Type": "text/html; charset=utf-8" }
    });
  }
};
