
export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const action = url.searchParams.get('api_action');

    // CORS Headers برای پاسخ به درخواست‌های برنامه شما
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Content-Type": "application/json"
    };

    if (action === 'validate') {
      const key = url.searchParams.get('key');
      const hwid = url.searchParams.get('hwid');

      if (!key || !hwid) {
        return new Response(JSON.stringify({ status: 'ERROR', message: 'پارامترهای ورودی ناقص است' }), { headers: corsHeaders });
      }

      // خواندن اطلاعات لایسنس از Cloudflare KV
      const licenseDataStr = await env.LICENSES_KV.get(key);
      if (!licenseDataStr) {
        return new Response(JSON.stringify({ status: 'INVALID', message: 'لایسنس یافت نشد' }), { headers: corsHeaders });
      }

      const license = JSON.parse(licenseDataStr);

      // بررسی تاریخ انقضا
      if (license.expire_date && new Date(license.expire_date) < new Date()) {
        return new Response(JSON.stringify({ status: 'EXPIRED', message: 'اعتبار لایسنس به پایان رسیده است' }), { headers: corsHeaders });
      }

      // مدیریت دستگاه‌ها در KV
      let devices = license.devices || [];
      if (devices.includes(hwid)) {
        // دستگاه قبلاً ثبت شده است
        return new Response(JSON.stringify({ status: 'SUCCESS', type: license.type, expires: license.expire_date }), { headers: corsHeaders });
      }

      // چک کردن سقف تعداد دستگاه‌ها
      if (devices.length >= license.max_devices) {
        return new Response(JSON.stringify({ status: 'MAX_DEVICES_REACHED', message: 'سقف تعداد دستگاه مجاز تکمیل شده است' }), { headers: corsHeaders });
      }

      // ثبت دستگاه جدید
      devices.push(hwid);
      license.devices = devices;
      await env.LICENSES_KV.put(key, JSON.stringify(license));

      return new Response(JSON.stringify({ status: 'SUCCESS', type: license.type, expires: license.expire_date }), { headers: corsHeaders });
    }

    return new Response(JSON.stringify({ status: 'ERROR', message: 'Invalid Action' }), { headers: corsHeaders });
  }
};
